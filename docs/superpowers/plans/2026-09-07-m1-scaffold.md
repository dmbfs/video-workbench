```ts
import type { FastifyReply } from "fastify";
import type { SseEvent } from "@vidstitch/shared";

const clients = new Map<string, Set<FastifyReply>>(); // projectId -> replies; "*" 全局

export function addClient(projectId: string | "*", reply: FastifyReply) {
  reply.raw.writeHead(200, { "Content-Type": "text/event-stream", "Cache-Control": "no-cache", Connection: "keep-alive" });
  if (!clients.has(projectId)) clients.set(projectId, new Set());
  clients.get(projectId)!.add(reply);
  reply.raw.on("close", () => clients.get(projectId)?.delete(reply));
}
export function broadcast(event: SseEvent, projectId = "*") {
  const payload = `data: ${JSON.stringify(event)}\n\n`;
  clients.get(projectId)?.forEach((r) => r.raw.write(payload));
  clients.get("*")?.forEach((r) => r.raw.write(payload));
}
```

```ts
export function getSettings(): Settings {
  const row = db.prepare("SELECT json FROM settings WHERE key='settings'").get() as { json: string } | undefined;
  return row ? settingsSchema.parse(JSON.parse(row.json)) : DEFAULTS;
}
export function saveSettings(s: Settings): Settings {
  const parsed = settingsSchema.parse(s);
  db.prepare("INSERT INTO settings(key,json) VALUES('settings',?) ON CONFLICT(key) DO UPDATE SET json=excluded.json")

  return parsed;
}
```

```ts
import { spawn } from "node:child_process";
import path from "node:path";
import ffmpegPath from "ffmpeg-static";
import { dataRoot } from "../db.js";
import type { VideoProvider, CreateTaskReq, CreateTaskCtx, PollResult, Caps } from "./types.js";

interface MockJob { child: ReturnType<typeof spawn>; out: string; code: number | null }
const jobs = new Map<string, MockJob>();

export class MockProvider implements VideoProvider {
  kind = "mock";
  capabilities(): Caps { return { maxSegmentDuration: 30, imageToVideo: true, audio: true }; }
  async createTask(req: CreateTaskReq, ctx: CreateTaskCtx) {
    const taskId = `mock_${ctx.segmentId}`;
    const dir = path.join(dataRoot, "projects", ctx.projectId, "segments");
    const out = path.join(dir, `${ctx.segmentId}.mp4`);
    const { mkdirSync } = await import("node:fs"); mkdirSync(dir, { recursive: true });
    const size = req.ratio === "16:9" ? "1280x720" : "720x1280";
    const hue = (ctx.idx * 67) % 360; // 不同段落不同色相，肉眼可辨拼接顺序
    const freq = 220 + ctx.idx * 60;  // 不同段落不同音调
    const child = spawn(ffmpegPath!, ["-y",
      "-f", "lavfi", "-i", `testsrc2=size=${size}:rate=30:duration=${req.duration}`,
      "-f", "lavfi", "-i", `sine=frequency=${freq}:duration=${req.duration}`,
      "-vf", `hue=h=${hue}:s=1.3,format=yuv420p`,
      "-c:v", "libx264", "-preset", "veryfast", "-c:a", "aac", "-ar", "48000", "-ac", "2",
      "-shortest", out]);
    jobs.set(taskId, { child, out, code: null });
    child.on("exit", (c) => { const j = jobs.get(taskId); if (j) j.code = c ?? 0; });
    return { taskId };
  }
  async pollTask(taskId: string): Promise<PollResult> {
    const j = jobs.get(taskId);
    if (!j) return { status: "failed", error: "unknown mock task" };
    if (j.code === null) return { status: "running" };
    return j.code === 0 ? { status: "succeeded", videoRef: j.out } : { status: "failed", error: "ffmpeg exit " + j.code };
  }
}
```

```ts
import type { VideoProvider, CreateTaskReq, CreateTaskCtx, PollResult, Caps } from "./types.js";
import type { ProviderConfig } from "@vidstitch/shared";

export class MiniMaxProvider implements VideoProvider {
  kind = "minimax";
  constructor(private cfg: ProviderConfig) {}
  capabilities(): Caps { return { maxSegmentDuration: 15, imageToVideo: true, audio: true }; }
  async createTask(req: CreateTaskReq, _ctx: CreateTaskCtx) {
    const content: object[] = [{ type: "text", text: req.prompt }];
    if (req.firstFrameB64) content.push({ type: "image_url", image_url: `data:image/jpeg;base64,${req.firstFrameB64}` });
    const r = await fetch(`${this.cfg.baseUrl}/v2/video_generation`, {
      method: "POST",
      headers: { "Content-Type": "application/json", Authorization: `Bearer ${this.cfg.apiKey}` },
      body: JSON.stringify({ model: this.cfg.modelId, content, duration: req.duration }),
    });
    if (!r.ok) throw new Error(`minimax create ${r.status}: ${(await r.text()).slice(0, 300)}`);
    const j = (await r.json()) as { task_id: string };
    return { taskId: j.task_id };
  }
  async pollTask(taskId: string): Promise<PollResult> {
    const r = await fetch(`${this.cfg.baseUrl}/v2/query/video_generation/${taskId}`, {
      headers: { Authorization: `Bearer ${this.cfg.apiKey}` } });
    if (!r.ok) return { status: "running" };
    const j = (await r.json()) as { status: string; content?: { url?: string } };
    if (j.status === "Success") return { status: "succeeded", videoRef: j.content!.url! };
    if (j.status === "Fail") return { status: "failed", error: "minimax task failed" };
    return { status: j.status === "Queue" ? "queued" : "running" };
  }
}
```

```ts
import { spawn } from "node:child_process";
import path from "node:path";
import { mkdtrIfMissing } from "./fsx.js";
import ffmpegPath from "ffmpeg-static";
import { dataRoot } from "./db.js";

const run = (args: string[]) => new Promise<void>((res, rej) => {
  const p = spawn(ffmpegPath!, args, { stdio: "pipe" });
  let err = ""; p.stderr.on("data", (d) => (err = (err + d).slice(-2000)));
  p.on("exit", (c) => (c === 0 ? res() : rej(new Error("ffmpeg: " + err))));
});

export async function stitch(projectId: string, segPaths: string[], ratio: "16:9" | "9:16",
  crossfadeMs: number, onPct: (pct: number, stage: "normalizing" | "concatenating" | "done") => void) {
  const size = ratio === "16:9" ? "1280x720" : "720x1280";
  const tmp = path.join(dataRoot, "projects", projectId, "tmp"); mkdtrIfMissing(tmp);
  const norm: string[] = [];
  for (let i = 0; i < segPaths.length; i++) {
    const o = path.join(tmp, `n${i}.mp4`);
    await run(["-y", "-i", segPaths[i],
      "-vf", `scale=${size.replace("x", ":")}:force_original_aspect_ratio=decrease,pad=${size}:(ow-iw)/2:(oh-ih)/2,fps=30,format=yuv420p`,
      "-c:v", "libx264", "-preset", "veryfast", "-c:a", "aac", "-ar", "48000", "-ac", "2", o]);
    norm.push(o); onPct(Math.round(((i + 1) / segPaths.length) * 80), "normalizing");
  }
  const list = path.join(tmp, "list.txt");
  const { writeFileSync } = await import("node:fs");
  writeFileSync(list, norm.map((p) => `file '${p.replace(/\\/g, "/")}'`).join("\n"));
  const out = path.join(dataRoot, "projects", projectId, "final.mp4");
  const vf = crossfadeMs > 0 ? ["-vf", `xfade=transition=fade:duration=${crossfadeMs / 1000}`] : [];
  await run(["-y", "-f", "concat", "-safe", "0", "-i", list, ...vf, "-c:v", crossfadeMs > 0 ? "libx264" : "copy",
    crossfadeMs > 0 ? "-preset" : "-c:a", ...(crossfadeMs > 0 ? ["veryfast", "-c:a", "aac"] : ["copy"]), out]);
  onPct(100, "done");
  return out;
}
```

```ts
import { EventEmitter } from "node:events";
import { writeFileSync } from "node:fs";
import path from "node:path";
import { db, dataRoot } from "./db.js";
import { getSettings } from "./settings.js";
import { getVideoProvider } from "./providers/factory.js";
import { broadcast } from "./sse.js";
import type { Segment } from "@vidstitch/shared";

class Orchestrator {
  private queue: string[] = []; private active = 0;
  concurrency = 2;
  enqueued = new EventEmitter(); // 内部唤醒
  constructor() { this.enqueued.on("wake", () => this.pump()); }
  enqueue(segmentId: string) {
    db.prepare("UPDATE segments SET status='pending', error=NULL WHERE id=?").run(segmentId);
    this.queue.push(segmentId); this.enqueued.emit("wake");
  }
  private seg(id: string) { return db.prepare("SELECT * FROM segments WHERE id=?").get(id) as Segment | undefined; }
  private setStatus(id: string, patch: Partial<Segment>) {
    const s = this.seg(id); if (!s) return;
    const next = { ...s, ...patch };
    db.prepare("UPDATE segments SET status=@status, provider=@provider, task_id=@taskId, video_path=@videoPath, error=@error WHERE id=@id").run({
      status: next.status, provider: next.provider ?? null, taskId: next.taskId ?? null,
      videoPath: next.videoPath ?? null, error: next.error ?? null, id });
    broadcast({ type: "segment_status", segmentId: id, status: next.status, error: next.error }, s.projectId);
  }
  private async pump() {
    while (this.active < this.concurrency && this.queue.length) {
      const id = this.queue.shift()!; this.active++;
      this.run(id).catch(() => {}).finally(() => { this.active--; this.enqueued.emit("wake"); });
    }
  }
  private async run(segmentId: string, attempt = 1) {
    const s = this.seg(segmentId); if (!s) return;
    const settings = getSettings();
    const cfg = settings.providers.find((p) => p.id === (s.provider ?? settings.videoDefaultId));
      ?? settings.providers.find((p) => p.kind === "mock")!;
    this.setStatus(segmentId, { status: "generating", provider: cfg.id, error: null });
    try {
      const provider = getVideoProvider(cfg);
      const { taskId } = await provider.createTask(
        { prompt: s.prompt, duration: s.duration, ratio: "16:9", withAudio: true }, // ratio 从 storyboard 读，见路由层传入
        { projectId: s.projectId, segmentId, idx: s.idx });
      this.setStatus(segmentId, { taskId });
      let videoRef: string | undefined;
      for (;;) {
        await new Promise((r) => setTimeout(r, 1500));
        const poll = await provider.pollTask(taskId);
        if (poll.status === "succeeded") { videoRef = poll.videoRef; break; }
        if (poll.status === "failed") throw new Error(poll.error ?? "provider failed");
      }
      const dest = path.join(dataRoot, "projects", s.projectId, "segments", `${segmentId}.mp4`);
      await resolveVideoRef(videoRef!, dest);
      this.setStatus(segmentId, { status: "succeeded", videoPath: dest });
    } catch (e) {
      const msg = (e as Error).message.slice(0, 300);
      if (attempt < 3) { await new Promise((r) => setTimeout(r, 2000 * attempt)); return this.run(segmentId, attempt + 1); }
      this.setStatus(segmentId, { status: "failed", error: msg });
    }
  }
}
async function resolveVideoRef(ref: string, dest: string) {
  if (/^https?:/.test(ref)) {
    const r = await fetch(ref); if (!r.ok) throw new Error("download " + r.status);
    writeFileSync(dest, Buffer.from(await r.arrayBuffer()));
  } else {
    const { copyFileSync } = await import("node:fs"); copyFileSync(ref, dest);
  }
}
export const orchestrator = new Orchestrator();
```

```ts
    db.prepare("DELETE FROM projects WHERE id=?").run(id);

    const { id } = req.params as any;
    const { prompt, duration } = addSegmentSchema.parse(req.body);
    const sid = newId();
    const idx = (db.prepare("SELECT COALESCE(MAX(idx),0)+1 n FROM segments WHERE project_id=?").get(id) as any).n;
    db.prepare("INSERT INTO segments(id,project_id,idx,prompt,duration) VALUES(?,?,?,?,?)").run(sid, id, idx, prompt, duration);
```
